```typescript
// Date helpers — stable relative dates regardless of when seed runs
function monthsAgo(n: number): Date {
  const d = new Date();
  d.setDate(1);
  d.setMonth(d.getMonth() - n);
  d.setHours(0, 0, 0, 0);
  return d;
}

function daysFromNow(n: number): Date {
  const d = new Date();
  d.setDate(d.getDate() + n);
  d.setHours(0, 0, 0, 0);
  return d;
}
```

```typescript
  const metricSites = [

  const avgRateMinor = 10500;

  for (let monthOffset = 5; monthOffset >= 0; monthOffset--) {
    const bucketAt = monthsAgo(monthOffset);

    for (const { site, baseOcc, baseAgreements } of metricSites) {
      // Occupancy grows slightly each month
      const occupancyPct = Math.min(92, baseOcc + (5 - monthOffset) * 1.8);
```
